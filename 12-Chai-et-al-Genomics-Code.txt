#Concatenating data from all chromosomes 
#----------------------------------------------------------------------
cat UKBF-EUR-Bmd-As-C-*.jma.cojo | grep -v SNP | awk '{ if ($1 == "25") {$1 = "23"}; print}' | sort -k1,1n -k3,3n | awk -F " " '{ OFS="\t" ; p = $10 ; q = 1-$10 ; b = $11^2 ; var = 2*p*q*b ; print $2,$1,$3,$10,$11,$12,$13,$9,$14,var}' > file1.txt

#Note: var = 2*p*q*b (BMD variance explained by each SNP) 

#Calculating distance between SNPs on same chr
#----------------------------------------------------------------------
cat file1.txt | awk -F "\t" '   BEGIN { OFS=FS ; prev2 = 0 ; prev3 = 0 } #2 = status (0/1) #3 = locus bin
                        {if ( $2 == prev2 )
                                { $11 = $3 - prev3 ;
                                print ;
                                prev2 = $2 ;
                                prev3 = $3 }
                        else if ( $2 != prev2 )
                                { $11 = -1 ;
                                print ;
                                prev2 = $2 ;
                                prev3 = $3 }
                        else
                                { $11 = "X" ;
                                print ;
                                prev2 = $2 ;
                                prev3 = $3 }
                        }' > file2.txt


#Define loci based on +/- 1mb sliding window
#----------------------------------------------------------------------
cat file2.txt | awk -F "\t" '   BEGIN {OFS=FS ; ind=1}
                        {if ( $11 < 1000000 && $11 > 0 )
                                { $12 = ind - 1 ;
                                print }
                         else if ( $11 > 1000000 || $11 < 0 )
                                { $12 = ++ind - 1 ;
                                print }
                         else
                                { $12 = "X" ;
                                print }
                         }' > file3.txt #Cojo-Summary Stats file

#Setting Flags:

input=commandArgs(trailingOnly=T)[1]

# Import sumstats
GWAS = data.frame(fread(input, sep=' ', header=T, colClasses=c(P="character"), na.strings="NA", stringsAsFactors=FALSE))

# Double precision for p-values, and convert to -log10(P)
.N <- function(.) mpfr(., precBits = 200)
GWAS$LP <- asNumeric(-log10(.N(GWAS$P)))

# sort by chromosome
unique(GWAS$CHR)
idx = sort(GWAS$CHR, index.return = T)$ix
GWAS = GWAS[idx,]

# get cumulative base pair positions and center positions
# thanks to Dani▒l Roelfs for his helpful blog post (http://www.danielroelfs.com/coding/manhattan_plots/)
nCHR = length(unique(GWAS$CHR))
GWAS$BPcum = NA
s = 0
nbp = c()
for (i in unique(GWAS$CHR)){
  nbp[i] = max(GWAS[GWAS$CHR == i,]$BP)
  GWAS[GWAS$CHR == i,"BPcum"] = GWAS[GWAS$CHR == i,"BP"] + s
  s = s + nbp[i]
}

axis.set = GWAS %>%
  group_by(CHR) %>%
  summarize(center = (max(BPcum) + min(BPcum)) / 2)

# set ylim, significance threshold, and threshold for labels
sig = 6.6E-9                    
sig_label = 8.18045606446  #Equivalent to -log10(6.6E-9)
#ylim = max(GWAS$LP)*1.10
ylim  = 600 # Set threshold due to outliers around WNT16-CPED1 Locus on Chr 7 with p < E-2000

# create manhattan plot 
set.seed(8245)
manhplot = ggplot(data = GWAS) + #ggplot
  geom_point(aes(x=BPcum, y=LP, color=as.factor(CHR)), alpha = 0.6, size = 2.5, stroke = 0, shape = 16) +
  scale_color_manual(values = rep(c("#6eca76", "#6e90ca"), nCHR)) +
  scale_x_continuous(expand = c(0.03,0), label = axis.set$CHR, breaks = axis.set$center) +
  scale_y_continuous(expand = c(0,0), limits = c(0,ylim), breaks = seq(0,ylim,100)) + #limits = c(0,ylim)
  geom_hline(yintercept = -log10(sig), color = "black", linetype = "dashed", size = 0.25) +
  labs(x = "Chromosome", y = expression("-log"[10]*"("*italic(p)*")")) +
  geom_segment(aes(x=min(axis.set$center),xend=max(axis.set$center),y=-Inf,yend=-Inf), colour = "grey70", size = 0.25)+
  geom_segment(aes(y=0,yend=ylim,x=-Inf,xend=-Inf), colour = "grey70", size = 0.25) +
  #Labelling novel loci
  geom_point(data=subset(GWAS, SAMPLE=="leadSNP" & STATUS=="NOVEL"), aes(x=BPcum, y=LP), fill="yellow", color="black", shape=21, size=2.5) +
  #Labelling cojo lead variants
  geom_text_repel(
    data          = subset(GWAS, SAMPLE=="leadSNP" & STATUS=="NOVEL"),
    aes(x=BPcum, y=LP, label=MAPPING), fontface = "italic",
    size          = 4,
    segment.size  = 0.1,
    min.segment.length = unit(0.02,"lines"),
    segment.color = "black",
    point.padding = unit(0, "lines"),
    direction     = "both",
    box.padding = 0.5,
    inherit.aes=FALSE,
    max.overlaps = Inf
  ) +
  theme_light(base_size = 12) +
  theme(
    legend.position = "none",
    panel.border = element_blank(),
    axis.line = element_blank(),
    axis.ticks.length=unit(.5, "cm"),
    panel.grid.major.x = element_blank(),
    panel.grid.minor.x = element_blank(),
    panel.grid.major.y = element_blank(),
    panel.grid.minor.y = element_blank(),
    axis.text.x = element_text(angle = 0, size = 8, vjust = 0.25, margin = margin(t = 3, r = 0, b = 0, l = 0)),
    axis.text.y = element_text(angle = 0, size = 8, vjust = 0.25, margin = margin(t = 0, r = 3, b = 0, l = 0)),
    axis.title.y = element_text(margin = margin(t = 0, r = 5, b = 0, l = 5)),
    axis.title.x = element_text(margin = margin(t = 10, r = 0, b = 0, l = 0)),
    plot.margin=unit(c(1.5,0.75,0.5,0),"cm")
  )

awk 'BEGIN {OFS="\t"} ($3 == 23) {$3 = "X"} 1' UKBF-EUR-Bmd-As-C-Hg38dbSnp151-Sumstats.txt  | awk 'BEGIN {OFS="\t" ; print "#CHROM","POS","ID","REF","ALT","QUAL","FILTER","INFO"} NR > 1 && $3 !=25 {print $3,$4,$2,$5,$6,"NA","PASS","SNPID="$1";AF="$7";INFO="$8";BETA="$9";SE="$10";P="$11";P.I="$12";P.NI="$13";N="$14}' > UKBF-EUR-Bmd-As-C-Hg38dbSnp151-Sumstats.vcf

#Run SnpEff Annotation
#----------------------------------------------------------
java -Xmx72g -jar snpEff.jar GRCh38.p14 -v UKBF-EUR-Bmd-As-C-Hg38dbSnp151-Sumstats.vcf > UKBF-EUR-Bmd-As-C-Hg38dbSnp151-Sumstats.ann

#Extract first annotation for each SNP
#----------------------------------------------------------
cat UKBF-EUR-Bmd-As-C-Hg38dbSnp151-Sumstats.ann | sed -e 's/|/\t/g' -e 's/;/\t/g' -e 's/=/\t/g' | awk -F "\t" 'BEGIN {OFS=FS ; print "#CHROM","BEGIN","END","MARKER_ID","RSID","EA","NEA","EAF","INFO","BETA","SE","PVALUE","N","IMPACT"} NR > 15 {print $1,$2,$2,$1":"$2"_"$4"/"$5"_"$28":"$30,$3,$4,$5,$11,$13,$15,$17,$23,$25,$29}' > UKBF-EUR-Bmd-As-C-Hg38dbSnp151-Sumstats-Epacts-22072021.txt

#Format and Qc GWAS Sumstats

awk -v chr=1 'BEGIN{print "SNP","CHR","BP","P","NOBS"} $3 == chr && $8 > 0.6 && $7 > 0.0005 && $7 < 0.9995 {print $2,$3,$4,$13,$14}' UKBF-EUR-Bmd-As-C-Hg38dbSnp151-Sumstats.txt > EUR-Bmd-As-C-Magma-SumStats-Chr1.txt

#Annotate Snps to Genes
#----------------------------------------------------------------------
magma.1.10 \
--annotate window=2,1 \
--snp-loc  UKBF-EUR-LdRefRsid-Hg38dbSnp151-Chr1.bim \
--gene-loc Ensemble.Protein.Coding.Genes.105.GRCh38.p13.Final.txt \
--out UKBF-EUR-LdRefRsid-Chr1-2u1dkb

#Run Gene-based Tests
#----------------------------------------------------------------------
magma.1.10 \
        --bfile UKBF-EUR-LdRefRsid-Hg38dbSnp151-Chr1 \
        --pval EUR-Bmd-As-C-Magma-SumStats-Chr1.txt ncol=NOBS \
        --gene-annot UKBF-EUR-LdRefRsid-Chr1-2u1dkb.genes.annot \
        --gene-settings snp-min-maf=0.0005 \
        --gene-model multi \
        --big-data \
        --out Magma-EUR-Bmd-As-C-Chr1-2u1dkb

rm(list=ls())
# load packages
library(data.table)
library(dplyr)
library(ggplot2)
library(ggrepel)
library(Rmpfr)

#Setting Flags: 
#----------------------------------------------------------
input=commandArgs(trailingOnly=T)[1]
output=commandArgs(trailingOnly=T)[2]

# Import sumstats
GWAS = data.frame(fread(input, sep='\t', header=T, colClasses=c(P_MULTI="character"), na.strings="NA", stringsAsFactors=FALSE))
head(GWAS)
# Double precision for p-values, and convert to -log10(P)
.N <- function(.) mpfr(., precBits = 200)
GWAS$LP <- asNumeric(-log10(.N(GWAS$P_MULTI)))

# sort by chromosome
unique(GWAS$CHR)
idx = sort(GWAS$CHR, index.return = T)$ix
GWAS = GWAS[idx,]

# get cumulative base pair positions and center positions
# thanks to Dani▒l Roelfs for his helpful blog post (http://www.danielroelfs.com/coding/manhattan_plots/)
nCHR = length(unique(GWAS$CHR))
GWAS$BPcum = NA
s = 0
nbp = c()
for (i in unique(GWAS$CHR)){
  nbp[i] = max(GWAS[GWAS$CHR == i,]$START)
  GWAS[GWAS$CHR == i,"BPcum"] = GWAS[GWAS$CHR == i,"START"] + s
  s = s + nbp[i]
}

axis.set = GWAS %>%
  group_by(CHR) %>%
  summarize(center = (max(BPcum) + min(BPcum)) / 2)

# set ylim, significance threshold, and threshold for labels
sig = 2.538715410002539e-6                      
sig_label = 5.59538598081  Equivalent to -log10(2.54E-6)
ylim = max(GWAS$LP)*1.05

# create manhattan plot #282873
set.seed(8245)
manhplot = ggplot(data = GWAS) + #ggplot
  geom_point(aes(x=BPcum, y=LP, color=as.factor(CHR)), alpha = 0.6, size = 1.5, stroke = 0, shape = 16) +
  scale_color_manual(values = rep(c("#6eca76", "#6e90ca"), nCHR)) +
  scale_x_continuous(expand = c(0.03,0), label = axis.set$CHR, breaks = axis.set$center) +
  scale_y_continuous(expand = c(0,0), limits = c(0,ylim), breaks = seq(0,ylim,10)) + #limits = c(0,ylim)
  #geom_hline(yintercept = -log10(sig), color = "black", linetype = "dashed", linewidth = 0.1) +
  labs(x = "", y = "") +
  geom_segment(aes(x=min(axis.set$center),xend=max(axis.set$center),y=-Inf,yend=-Inf), colour = "grey70", linewidth = 0.25)+
  geom_segment(aes(y=0,yend=ylim,x=-Inf,xend=-Inf), colour = "grey70", linewidth = 0.25) +
  #Labelling novel loci
  geom_point(data=subset(GWAS, NOSOLOGY==1 & GWAS$LP > sig_label), aes(x=BPcum, y=LP), fill="orange", color="black", shape=21, size=1.5) +
  #Labelling cojo lead variants
  geom_text_repel(
    data          = subset(GWAS, NOSOLOGY==1 & GWAS$LP > sig_label),
    aes(x=BPcum, y=LP, label=GENE.SYMBOL), fontface = "italic",
    size          = 2,
    segment.size  = 0.1,
    #min.segment.length = unit(0.02,"lines"),
    segment.color = "black",
    #point.padding = unit(0, "lines"),
    direction     = "both",
    #box.padding = 1,
    inherit.aes=FALSE,
    max.overlaps = Inf
  ) +
  theme_light(base_size = 6) +
  theme(
    legend.position = "none",
    panel.border = element_blank(),
    axis.line = element_blank(),
    axis.ticks.length=unit(.1, "cm"),
    panel.grid.major.x = element_blank(),
    panel.grid.minor.x = element_blank(),
    panel.grid.major.y = element_blank(),
    panel.grid.minor.y = element_blank(),
    #axis.text.x = element_text(angle = 0, vjust = 0.25, margin = margin(t = 3, r = 0, b = 0, l = 0)),
    #axis.text.y = element_text(angle = 0, vjust = 0.25, margin = margin(t = 0, r = 3, b = 0, l = 0)),
    #axis.title.y = element_text(margin = margin(t = 0, r = 5, b = 0, l = 5)),
    #axis.title.x = element_text(margin = margin(t = 10, r = 0, b = 0, l = 0)),
    #plot.margin=unit(c(1.5,0.75,0.5,0),"cm")
  )

# draw manhattan plot
ggsave(file=paste(output,"png",sep="."), manhplot, width = 17.9, height = 6.3, unit="cm", dpi = 300)
